<script>
    // Variables globales para el wizard
    let currentStep = 1;
    const totalSteps = 3;

    // Referencias a elementos del DOM
    const formRegistroClientePersona = document.getElementById('form-registro-cliente-persona');
    const btnSiguiente = document.getElementById('btn-siguiente');
    const btnAnterior = document.getElementById('btn-anterior');
    const btnRegistrar = document.getElementById('btn-registrar');

    // Función para mostrar un paso específico
    function showStep(step) {
        // Ocultar todos los pasos
        document.querySelectorAll('.form-step').forEach(stepElement => {
            stepElement.classList.remove('active');
        });
        
        // Mostrar el paso actual
        const currentStepElement = document.getElementById(`step-${step}`);
        if (currentStepElement) {
            currentStepElement.classList.add('active');
        }
        
        // Actualizar botones
        updateButtons();
        
        // Si es el paso 3, generar resumen
        if (step === 3) {
            generateSummary();
        }
    }

    // Función para actualizar la visibilidad de los botones
    function updateButtons() {
        // Botón anterior
        if (currentStep === 1) {
            btnAnterior.style.display = 'none';
        } else {
            btnAnterior.style.display = 'inline-block';
        }
        
        // Botón siguiente y registrar
        if (currentStep === totalSteps) {
            btnSiguiente.style.display = 'none';
            btnRegistrar.style.display = 'inline-block';
        } else {
            btnSiguiente.style.display = 'inline-block';
            btnRegistrar.style.display = 'none';
        }
    }

    // Función para validar el paso actual
    function validateCurrentStep() {
        const currentStepElement = document.getElementById(`step-${currentStep}`);
        const requiredFields = currentStepElement.querySelectorAll('[required]');
        let isValid = true;
        
        requiredFields.forEach(field => {
            // Limpiar clases previas
            field.classList.remove('is-valid', 'is-invalid');
            
            if (!field.value.trim()) {
                field.classList.add('is-invalid');
                isValid = false;
            } else {
                field.classList.add('is-valid');
            }
        });
        
        // Validaciones específicas por paso
        if (currentStep === 1) {
            // Validar documento
            const tipoDoc = document.getElementById('tipodocumento').value;
            const nroDoc = document.getElementById('ndocumento').value;
            
            if (tipoDoc === 'DNI' && nroDoc.length !== 8) {
                document.getElementById('ndocumento').classList.add('is-invalid');
                isValid = false;
            }
            
            // Validar fecha de nacimiento
            const fechaNac = document.getElementById('fechanac').value;
            if (fechaNac) {
                const today = new Date();
                const birthDate = new Date(fechaNac);
                const age = today.getFullYear() - birthDate.getFullYear();
                
                if (age < 18 || age > 100) {
                    document.getElementById('fechanac').classList.add('is-invalid');
                    isValid = false;
                }
            }
        }
        
        if (currentStep === 2) {
            // Validar teléfono
            const telefono = document.getElementById('telprimario').value;
            if (telefono && telefono.length !== 9) {
                document.getElementById('telprimario').classList.add('is-invalid');
                isValid = false;
            }
        }
        
        return isValid;
    }

    // Función para generar el resumen
    function generateSummary() {
        const resumenContainer = document.getElementById('resumen-datos');
        
        const datos = {
            'Documento': `${document.getElementById('tipodocumento').value} - ${document.getElementById('ndocumento').value}`,
            'Nombres': `${document.getElementById('apellidos').value}, ${document.getElementById('nombres').value}`,
            'Género': document.getElementById('genero').selectedOptions[0]?.text || '',
            'Fecha de nacimiento': document.getElementById('fechanac').value,
            'Estado civil': document.getElementById('estadocivil').selectedOptions[0]?.text || 'No especificado',
            'Ubicación': `${document.getElementById('departamento').selectedOptions[0]?.text || ''} - ${document.getElementById('provincia').selectedOptions[0]?.text || ''} - ${document.getElementById('distrito').selectedOptions[0]?.text || ''}`,
            'Dirección': document.getElementById('direccion').value || 'No especificada',
            'Teléfono': document.getElementById('telprimario').value,
            'Email': document.getElementById('email').value || 'No especificado'
        };
        
        let html = '<div class="row g-2">';
        for (const [key, value] of Object.entries(datos)) {
            if (value && value.trim()) {
                html += `
                    <div class="col-md-6">
                        <strong>${key}:</strong> ${value}
                    </div>
                `;
            }
        }
        html += '</div>';
        
        resumenContainer.innerHTML = html;
    }

    // Event listeners
    btnSiguiente.addEventListener('click', () => {
        if (validateCurrentStep()) {
            if (currentStep < totalSteps) {
                currentStep++;
                showStep(currentStep);
            }
        } else {
            // Mostrar mensaje de error
            showToast('Por favor, complete todos los campos obligatorios correctamente.', 'ERROR', 1300);
        }
    });

    btnAnterior.addEventListener('click', () => {
        if (currentStep > 1) {
            currentStep--;
            showStep(currentStep);
        }
    });

    // Validación en tiempo real
    document.addEventListener('input', (e) => {
        if (e.target.matches('input[required], select[required]')) { // matchea los campos que tengan el atributo required
            const field = e.target;
            field.classList.remove('is-valid', 'is-invalid');
            
            if (field.value.trim()) {
                field.classList.add('is-valid');
            }
        }
    });

    // Inicializar el wizard
    document.addEventListener('DOMContentLoaded', () => {
        showStep(1);
    });

    // Manejo del envío del formulario
    formRegistroClientePersona.addEventListener('submit', async(event) => {
        event.preventDefault();

        // Validar todos los pasos
        let allValid = true;
        for (let step = 1; step <= totalSteps; step++) {
            const tempCurrentStep = currentStep;
            currentStep = step;
            if (!validateCurrentStep()) {
                allValid = false;
                showStep(step); // Ir al primer paso con errores
                break;
            }
            currentStep = tempCurrentStep;
        }

        if (allValid) {
            if (await ask("¿Desea registrar este cliente?", "Registrar cliente")) {
                event.target.submit();
            }
        } else {
            showToast('Por favor, complete todos los campos obligatorios en todos los pasos.', 'ERROR', 1300);
        }
    });

    // Búsqueda de cliente por documento
    document.getElementById('btnBuscarCliente').addEventListener('click', async () => {
        const tipoDoc = document.getElementById('tipodocumento').value;
        const nroDoc = document.getElementById('ndocumento').value.trim();

        if (!nroDoc) {
            showToast('Ingrese un número de documento', 'ERROR', 1300);
            return;
        }

        if (tipoDoc === 'DNI' && nroDoc.length !== 8) {
            showToast('El DNI debe tener 8 dígitos', 'ERROR', 1300);
            return;
        }

        try {
            // Mostrar loading
            const btn = document.getElementById('btnBuscarCliente');
            const originalHTML = btn.innerHTML;
            btn.innerHTML = '<i class="bi bi-hourglass-split"></i>';
            btn.disabled = true;

            // Buscar en base de datos local primero
            const localResponse = await fetch(`/api/clientes/buscar/${nroDoc}`);
            
            if (localResponse.ok) {
                const localData = await localResponse.json();
                if (localData.exists) {
                    showToast('Este cliente ya está registrado', 'ERROR', 1300);
                    btn.innerHTML = originalHTML;
                    btn.disabled = false;
                    return;
                }
            }

            // Si no existe localmente, buscar en RENIEC (solo para DNI)
            if (tipoDoc === 'DNI') {
                const reniecResponse = await fetch(`/api/reniec/${nroDoc}`);
                
                if (reniecResponse.ok) {
                    const reniecData = await reniecResponse.json();
                    
                    if (reniecData.success) {
                        // Llenar campos con datos de RENIEC
                        document.getElementById('nombres').value = reniecData.nombres || '';
                        document.getElementById('apellidos').value = `${reniecData.apellidoPaterno || ''} ${reniecData.apellidoMaterno || ''}`.trim();
                        
                        showToast('Datos obtenidos de RENIEC', 'SUCCESS', 1300);
                    } else {
                        showToast('No se encontraron datos en RENIEC', 'WARNING', 1300);
                    }
                } else {
                    showToast('Error al consultar RENIEC', 'ERROR', 1300);
                }
            }

        } catch (error) {
            console.error('Error:', error);
            showToast('Error en la búsqueda', 'ERROR', 1300);
        } finally {
            // Restaurar botón
            const btn = document.getElementById('btnBuscarCliente');
            btn.innerHTML = '<i class="bi bi-search"></i>';
            btn.disabled = false;
        }
    });

    // Validación de solo números para teléfonos
    document.querySelectorAll('#telprimario, #telsecundario').forEach(input => {
        input.addEventListener('input', function(e) {
            this.value = this.value.replace(/\D/g, '');
        });
    });

    // Validación de solo números para documento
    document.getElementById('ndocumento').addEventListener('input', function(e) {
        const tipoDoc = document.getElementById('tipodocumento').value;
        if (tipoDoc === 'DNI') {
            this.value = this.value.replace(/\D/g, '');
            if (this.value.length > 8) {
                this.value = this.value.slice(0, 8);
            }
        }
    });

    // Cambio de tipo de documento
    document.getElementById('tipodocumento').addEventListener('change', function(e) {
        const nroDocInput = document.getElementById('ndocumento');
        nroDocInput.value = '';
        
        if (e.target.value === 'DNI') {
            nroDocInput.maxLength = 8;
            nroDocInput.placeholder = 'Ingrese 8 dígitos';
        } else {
            nroDocInput.maxLength = 12;
            nroDocInput.placeholder = 'Ingrese el número';
        }
    });
</script>
